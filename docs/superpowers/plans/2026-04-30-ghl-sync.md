# GHL Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bidirectional sync between Avantej Core and GoHighLevel — outbound CRUD on customers/staff mirrors to GHL Contacts/Users; inbound webhooks create new customers from GHL form submissions while ignoring updates/deletes on existing customers (Avantej is canonical).

**Architecture:** Inline synchronous outbound calls per CRUD mutation (failure mode B: Avantej write succeeds, GHL failure flagged for retry). Per-branch GHL location with per-branch Private Integration API key (encrypted at rest). Source of truth: Avantej. Cardinality: 1 Avantej branch = 1 GHL sub-account/location.

**Tech Stack:** Drizzle ORM (Postgres), bun:test (unit + integration), oRPC (typed RPC), TanStack Router (web), Cloudflare Workers (server runtime), Neon (Postgres-as-a-service). Existing GHL infrastructure: `gohighlevel.service.ts`, `webhooks.routes.ts`, `ghl_*` schema tables.

**Source spec:** `docs/superpowers/specs/2026-04-30-ghl-sync-design.md` (commit `2301aad`).

// packages/db/src/schema/control-panel/ghl-branch-links.ts
/**
 * Control Panel GoHighLevel Branch Links Schema
 *
 * Maps Avantej branches to GoHighLevel sub-account/location IDs and stores
 * the per-branch Private Integration API key (encrypted) used for outbound
 * sync calls and the webhook secret used for inbound HMAC verification.
 *
 * Cardinality: one row per (organization, branch). One GHL location can
 * only be linked to one Avantej branch (enforced by UNIQUE on
 * ghl_location_id).
 */

import {
	boolean,
	index,
	pgTable,
	text,
	timestamp,
	uuid,
	uniqueIndex,
} from "drizzle-orm/pg-core";
import { organizations } from "./organizations";

export const ghlBranchLinks = pgTable(
	"ghl_branch_links",
	{
		id: uuid("id").primaryKey().defaultRandom(),
		organizationId: uuid("organization_id")
			.notNull()
			.references(() => organizations.id, { onDelete: "cascade" }),
		// Cross-DB ref to hierarchy_nodes.id where type='branch' (no FK)
		tenantBranchId: uuid("tenant_branch_id").notNull(),
		ghlLocationId: text("ghl_location_id").notNull(),
		// Encrypted via the same ENCRYPTION_KEY pattern as
		// organizations.connection_string. Plaintext value never persisted.
		ghlApiKey: text("ghl_api_key").notNull(),
		// Encrypted webhook secret for inbound HMAC verification.
		webhookSecret: text("webhook_secret"),
		isActive: boolean("is_active").default(true).notNull(),
		createdAt: timestamp("created_at").defaultNow().notNull(),
		updatedAt: timestamp("updated_at").defaultNow().notNull(),
	},
	(table) => ({
		branchUniqueIdx: uniqueIndex("ghl_branch_links_org_branch_unique").on(
			table.organizationId,
			table.tenantBranchId,
		),
		ghlLocationUniqueIdx: uniqueIndex(
			"ghl_branch_links_ghl_location_unique",
		).on(table.ghlLocationId),
		organizationIdIdx: index("ghl_branch_links_org_idx").on(
			table.organizationId,
		),
		tenantBranchIdIdx: index("ghl_branch_links_branch_idx").on(
			table.tenantBranchId,
		),
		isActiveIdx: index("ghl_branch_links_active_idx").on(table.isActive),
	}),
);
```

// packages/db/src/schema/tenant/ghl-contact-links.ts
/**
 * Tenant GoHighLevel Contact Links Schema
 *
 * Maps GoHighLevel contacts to internal customer profile records.
 * Each (customer, branch) pair maps to exactly one GHL Contact in the
 * branch's GHL location (cardinality enforced by UNIQUE).
 */

import {
	index,
	pgTable,
	text,
	timestamp,
	uuid,
	uniqueIndex,
} from "drizzle-orm/pg-core";
import { customerProfiles } from "./customer-profiles";

export const ghlContactLinks = pgTable(
	"ghl_contact_links",
	{
		id: uuid("id").primaryKey().defaultRandom(),
		ghlContactId: text("ghl_contact_id").notNull(),
		userId: text("user_id").notNull(),
		customerProfileId: uuid("customer_profile_id").references(
			() => customerProfiles.id,
			{ onDelete: "set null" },
		),
		// NEW: which Avantej branch this contact belongs to.
		// Cross-DB ref to hierarchy_nodes.id where type='branch' (no FK).
		branchNodeId: uuid("branch_node_id").notNull(),
		// NEW: denormalized ghl_location_id for outbound calls without a
		// control-panel join.
		ghlLocationId: text("ghl_location_id").notNull(),
		// NEW: last GHL sync error (failure mode B observability).
		lastError: text("last_error"),
		syncStatus: text("sync_status").default("active").notNull(),
		lastSyncedAt: timestamp("last_synced_at"),
		createdAt: timestamp("created_at").defaultNow().notNull(),
		updatedAt: timestamp("updated_at").defaultNow().notNull(),
	},
	(table) => ({
		ghlContactIdUniqueIdx: uniqueIndex(
			"ghl_contact_links_ghl_contact_id_unique",
		).on(table.ghlContactId),
		// NEW: unique on (customer, branch) — same customer at multiple
		// branches has multiple Contact rows.
		customerBranchUniqueIdx: uniqueIndex(
			"ghl_contact_links_customer_branch_unique",
		).on(table.customerProfileId, table.branchNodeId),
		userIdIdx: index("ghl_contact_links_user_idx").on(table.userId),
		customerProfileIdIdx: index("ghl_contact_links_customer_profile_idx").on(
			table.customerProfileId,
		),
		branchNodeIdIdx: index("ghl_contact_links_branch_idx").on(
			table.branchNodeId,
		),
		ghlLocationIdIdx: index("ghl_contact_links_ghl_location_idx").on(
			table.ghlLocationId,
		),
		syncStatusIdx: index("ghl_contact_links_sync_status_idx").on(
			table.syncStatus,
			table.lastSyncedAt,
		),
		lastSyncedAtIdx: index("ghl_contact_links_last_synced_idx").on(
			table.lastSyncedAt,
		),
	}),
);
```

// apps/server/src/services/ghl/http-client.ts
/**
 * GoHighLevel HTTP client.
 *
 * Wraps fetch with:
 *  - GHL base URL (https://services.leadconnectorhq.com)
 *  - Auth header (Bearer + Version 2021-07-28)
 *  - Timeout (default 5s)
 *  - Error normalization to GhlApiError
 *  - Optional query params for GET
 */

import { GhlApiError, type GhlApiErrorCode } from "./types";

const GHL_BASE_URL = "https://services.leadconnectorhq.com";
const GHL_API_VERSION = "2021-07-28";
const DEFAULT_TIMEOUT_MS = 5000;

export interface GhlFetchArgs {
	method: "GET" | "POST" | "PUT" | "DELETE" | "PATCH";
	path: string;
	apiKey: string;
	body?: unknown;
	query?: Record<string, string | number | undefined>;
	timeoutMs?: number;
}

export async function ghlFetch<T = unknown>(
	args: GhlFetchArgs,
): Promise<T> {
	const { method, path, apiKey, body, query, timeoutMs = DEFAULT_TIMEOUT_MS } =
		args;

	let url = `${GHL_BASE_URL}${path}`;
	if (query) {
		const params = new URLSearchParams();
		for (const [k, v] of Object.entries(query)) {
			if (v !== undefined && v !== null) {
				params.append(k, String(v));
			}
		}
		const qs = params.toString();
		if (qs) url += `?${qs}`;
	}

	const controller = new AbortController();
	const timeoutId = setTimeout(() => controller.abort(), timeoutMs);

	let response: Response;
	try {
		response = await fetch(url, {
			method,
			headers: {
				Authorization: `Bearer ${apiKey}`,
				Version: GHL_API_VERSION,
				"Content-Type": "application/json",
				Accept: "application/json",
			},
			body: body ? JSON.stringify(body) : undefined,
			signal: controller.signal,
		});
	} catch (error) {
		clearTimeout(timeoutId);
		if (error instanceof Error && error.name === "AbortError") {
			throw new GhlApiError(
				"timeout",
				undefined,
				`Request to ${path} timed out after ${timeoutMs}ms`,
			);
		}
		throw new GhlApiError(
			"network",
			undefined,
			`Network error calling ${path}: ${error instanceof Error ? error.message : String(error)}`,
		);
	}
	clearTimeout(timeoutId);

	if (response.ok) {
		const text = await response.text();
		if (!text) return undefined as T;
		try {
			return JSON.parse(text) as T;
		} catch {
			throw new GhlApiError(
				"unknown",
				response.status,
				`Failed to parse JSON response from ${path}`,
			);
		}
	}

	const code: GhlApiErrorCode =
		response.status === 401
			? "unauthorized"
			: response.status === 404
				? "not_found"
				: response.status === 429
					? "rate_limited"
					: response.status >= 400 && response.status < 500
						? "validation"
						: "unknown";

	let bodyText: unknown = undefined;
	try {
		bodyText = await response.json();
	} catch {
		bodyText = await response.text().catch(() => "");
	}

	throw new GhlApiError(
		code,
		response.status,
		`GHL API ${response.status} on ${path}`,
		bodyText,
	);
}
```

export function buildContactPayload(
	args: BuildContactArgs,
): GhlContactPayload {
	const { locationId, user, profile, memberStatus, customFieldIds } = args;

	const tags: string[] = [];
	tags.push(memberStatus === "active" ? "customer:active" : "customer:inactive");
	if (profile.fitnessGoals) {
		for (const g of profile.fitnessGoals) tags.push(`goal:${g}`);
	}

	const customFields: Array<{ id: string; value: string | number }> = [];
	const addCustomField = (
		key: CustomFieldKey,
		value: string | number | null,
	) => {
		const id = customFieldIds[key];
		if (id && value !== null && value !== "") {
			customFields.push({ id, value });
		}
	};
	addCustomField("gender", profile.gender);
	addCustomField("emergencyContactName", profile.emergencyContactName);
	addCustomField("emergencyContactPhone", profile.emergencyContactPhone);
	addCustomField("medicalNotes", profile.medicalNotes);
	addCustomField("creditPoints", profile.creditPoints);
	addCustomField("notes", profile.notes);

	return {
		locationId,
		email: user.email,
		firstName: user.firstName ?? undefined,
		lastName: user.lastName ?? undefined,
		name: user.name ?? undefined,
		phone: profile.phone ?? undefined,
		address1: profile.address ?? undefined,
		dateOfBirth: profile.dateOfBirth
			? profile.dateOfBirth.toISOString().split("T")[0]
			: undefined,
		tags,
		customFields,
		source: "Avantej Sync",
	};
}

// apps/server/src/services/ghl/setup.ts
/**
 * Setup utilities for connecting a new GHL location:
 *  - validateGhlApiKey: confirms an API key works for the given locationId
 *  - ensureCustomFieldsExist: idempotent custom-field provisioning
 */

import { ghlFetch } from "./http-client";
import type { CustomFieldIdMap, CustomFieldKey } from "./payload-builders";
import type { GhlCustomField, GhlLocation } from "./types";
import { GhlApiError } from "./types";

export async function validateGhlApiKey(
	apiKey: string,
	locationId: string,
): Promise<boolean> {
	try {
		const loc = await ghlFetch<GhlLocation>({
			method: "GET",
			path: `/locations/${locationId}`,
			apiKey,
		});
		return loc.id === locationId;
	} catch (error) {
		if (error instanceof GhlApiError) return false;
		throw error;
	}
}

interface CustomFieldDef {
	key: CustomFieldKey;
	name: string;
	fieldKey: string;
	dataType: GhlCustomField["dataType"];
}

const REQUIRED_CUSTOM_FIELDS: CustomFieldDef[] = [
	{ key: "gender", name: "Gender", fieldKey: "gender", dataType: "TEXT" },
	{
		key: "emergencyContactName",
		name: "Emergency Contact Name",
		fieldKey: "emergency_contact_name",
		dataType: "TEXT",
	},
	{
		key: "emergencyContactPhone",
		name: "Emergency Contact Phone",
		fieldKey: "emergency_contact_phone",
		dataType: "PHONE",
	},
	{
		key: "medicalNotes",
		name: "Medical Notes",
		fieldKey: "medical_notes",
		dataType: "LARGE_TEXT",
	},
	{
		key: "creditPoints",
		name: "Credit Points",
		fieldKey: "credit_points",
		dataType: "NUMERICAL",
	},
	{
		key: "notes",
		name: "Internal Notes",
		fieldKey: "internal_notes",
		dataType: "LARGE_TEXT",
	},
];

export async function ensureCustomFieldsExist(args: {
	apiKey: string;
	locationId: string;
}): Promise<CustomFieldIdMap> {
	const { apiKey, locationId } = args;

	const existing = await ghlFetch<{ customFields: GhlCustomField[] }>({
		method: "GET",
		path: `/locations/${locationId}/customFields`,
		apiKey,
	});

	const byFieldKey = new Map<string, string>();
	for (const cf of existing.customFields) {
		byFieldKey.set(cf.fieldKey, cf.id);
	}

	const result: CustomFieldIdMap = {};
	for (const def of REQUIRED_CUSTOM_FIELDS) {
		const existingId = byFieldKey.get(def.fieldKey);
		if (existingId) {
			result[def.key] = existingId;
			continue;
		}
		const created = await ghlFetch<{ id: string }>({
			method: "POST",
			path: `/locations/${locationId}/customFields`,
			apiKey,
			body: {
				name: def.name,
				fieldKey: def.fieldKey,
				dataType: def.dataType,
				model: "contact",
			},
		});
		result[def.key] = created.id;
	}
	return result;
}

// apps/server/src/services/ghl/sync-customer.ts
/**
 * Customer (member.role='client' + customer_profiles + user) → GHL Contact
 * outbound sync orchestrator.
 *
 * Failure mode B: Avantej caller's mutation has already committed before
 * this is invoked. We attempt the GHL call best-effort and either:
 *   - synced  → upsert ghl_contact_links { sync_status='active', ... }
 *   - skipped → no link, no GHL config
 *   - error   → upsert ghl_contact_links { sync_status='error', last_error }
 *
 * The caller never sees a sync error as a thrown exception.
 */

import { and, eq } from "drizzle-orm";
import { ghlBranchLinks } from "@avantej-core/db/schema/control-panel/ghl-branch-links";
import { ghlContactLinks } from "@avantej-core/db/schema/tenant/ghl-contact-links";
import type { NeonDatabase } from "drizzle-orm/neon-serverless";
import type * as controlPanelSchema from "@avantej-core/db/schema/control-panel-schema";
import type * as tenantSchema from "@avantej-core/db/schema/tenant-schema";
import { ghlFetch } from "./http-client";
import {
	buildContactPayload,
	type AvantejUser,
	type AvantejCustomerProfile,
	type CustomFieldIdMap,
} from "./payload-builders";
import type { GhlContact } from "./types";
import { GhlApiError } from "./types";

export type SyncCustomerOperation = "create" | "update" | "delete";

export interface SyncCustomerArgs {
	controlPanelDb: NeonDatabase<typeof controlPanelSchema>;
	tenantDb: NeonDatabase<typeof tenantSchema>;
	organizationId: string;
	branchNodeId: string;
	customerProfile: AvantejCustomerProfile;
	user: AvantejUser;
	memberStatus: "active" | "inactive";
	operation: SyncCustomerOperation;
	/**
	 * Decrypt function for the stored ghl_api_key. Injected to avoid
	 * making this module depend on the encryption util directly (easier
	 * unit testing).
	 */
	decryptApiKey: (encrypted: string) => string;
	/**
	 * Custom field IDs resolved per location. Caller should fetch these
	 * once per location and cache. If empty, custom fields are skipped
	 * (still functional for basic name/email/phone sync).
	 */
	customFieldIds?: CustomFieldIdMap;
}

export interface SyncCustomerResult {
	status: "synced" | "skipped" | "error";
	ghlContactId?: string;
	error?: string;
	reason?: "branch-not-connected" | "no-existing-contact-on-update";
}

export async function syncCustomerToGhl(
	args: SyncCustomerArgs,
): Promise<SyncCustomerResult> {
	const {
		controlPanelDb,
		tenantDb,
		organizationId,
		branchNodeId,
		customerProfile,
		user,
		memberStatus,
		operation,
		decryptApiKey,
		customFieldIds = {},
	} = args;

	// 1. Resolve the branch's GHL connection
	const link = await controlPanelDb.query.ghlBranchLinks.findFirst({
		where: and(
			eq(ghlBranchLinks.organizationId, organizationId),
			eq(ghlBranchLinks.tenantBranchId, branchNodeId),
			eq(ghlBranchLinks.isActive, true),
		),
	});

	if (!link) {
		return { status: "skipped", reason: "branch-not-connected" };
	}

	const apiKey = decryptApiKey(link.ghlApiKey);
	const locationId = link.ghlLocationId;

	// 2. Look up existing contact link for this (customer, branch)
	const existing = await tenantDb.query.ghlContactLinks.findFirst({
		where: and(
			eq(ghlContactLinks.customerProfileId, customerProfile.id),
			eq(ghlContactLinks.branchNodeId, branchNodeId),
		),
	});

	// 3. Dispatch by operation
	try {
		if (operation === "create") {
			const payload = buildContactPayload({
				locationId,
				user,
				profile: customerProfile,
				memberStatus,
				customFieldIds,
			});
			const created = await ghlFetch<GhlContact>({
				method: "POST",
				path: "/contacts/",
				apiKey,
				body: payload,
			});
			await upsertLink({
				tenantDb,
				ghlContactId: created.id,
				userId: user.id,
				customerProfileId: customerProfile.id,
				branchNodeId,
				ghlLocationId: locationId,
				syncStatus: "active",
				lastError: null,
			});
			return { status: "synced", ghlContactId: created.id };
		}

		if (operation === "update") {
			if (!existing) {
				// No prior sync — treat as create
				const payload = buildContactPayload({
					locationId,
					user,
					profile: customerProfile,
					memberStatus,
					customFieldIds,
				});
				const created = await ghlFetch<GhlContact>({
					method: "POST",
					path: "/contacts/",
					apiKey,
					body: payload,
				});
				await upsertLink({
					tenantDb,
					ghlContactId: created.id,
					userId: user.id,
					customerProfileId: customerProfile.id,
					branchNodeId,
					ghlLocationId: locationId,
					syncStatus: "active",
					lastError: null,
				});
				return { status: "synced", ghlContactId: created.id };
			}
			const payload = buildContactPayload({
				locationId,
				user,
				profile: customerProfile,
				memberStatus,
				customFieldIds,
			});
			await ghlFetch<GhlContact>({
				method: "PUT",
				path: `/contacts/${existing.ghlContactId}`,
				apiKey,
				body: payload,
			});
			await upsertLink({
				tenantDb,
				ghlContactId: existing.ghlContactId,
				userId: user.id,
				customerProfileId: customerProfile.id,
				branchNodeId,
				ghlLocationId: locationId,
				syncStatus: "active",
				lastError: null,
			});
			return { status: "synced", ghlContactId: existing.ghlContactId };
		}

		// operation === "delete"
		if (!existing) {
			return { status: "skipped", reason: "no-existing-contact-on-update" };
		}
		await ghlFetch<unknown>({
			method: "DELETE",
			path: `/contacts/${existing.ghlContactId}`,
			apiKey,
		});
		await upsertLink({
			tenantDb,
			ghlContactId: existing.ghlContactId,
			userId: user.id,
			customerProfileId: customerProfile.id,
			branchNodeId,
			ghlLocationId: locationId,
			syncStatus: "deleted",
			lastError: null,
		});
		return { status: "synced", ghlContactId: existing.ghlContactId };
	} catch (error) {
		const message =
			error instanceof GhlApiError
				? `${error.code}: ${error.message}`
				: error instanceof Error
					? error.message
					: String(error);
		await upsertLink({
			tenantDb,
			ghlContactId: existing?.ghlContactId ?? "pending",
			userId: user.id,
			customerProfileId: customerProfile.id,
			branchNodeId,
			ghlLocationId: locationId,
			syncStatus: "error",
			lastError: message,
		});
		return { status: "error", error: message };
	}
}

async function upsertLink(args: {
	tenantDb: NeonDatabase<typeof tenantSchema>;
	ghlContactId: string;
	userId: string;
	customerProfileId: string;

// apps/server/src/services/ghl/sync-staff.ts
/**
 * Staff (member.role IN admin/staff/trainer + user) → GHL User outbound sync.
 *
 * Same failure mode B as sync-customer.ts. Different table: ghl_user_links.
 */

import { and, eq } from "drizzle-orm";
import { ghlBranchLinks } from "@avantej-core/db/schema/control-panel/ghl-branch-links";
import { ghlUserLinks } from "@avantej-core/db/schema/tenant/ghl-user-links";
import type { NeonDatabase } from "drizzle-orm/neon-serverless";
import type * as controlPanelSchema from "@avantej-core/db/schema/control-panel-schema";
import type * as tenantSchema from "@avantej-core/db/schema/tenant-schema";
import { ghlFetch } from "./http-client";
import { buildUserPayload, type AvantejUser } from "./payload-builders";
import type { GhlUser } from "./types";
import { GhlApiError } from "./types";

export interface AvantejMember {
	id: string;
	userId: string;
	role: string;
	status: string;
	hierarchyScopeId: string;
}

export type SyncStaffOperation = "create" | "update" | "delete";

export interface SyncStaffArgs {
	controlPanelDb: NeonDatabase<typeof controlPanelSchema>;
	tenantDb: NeonDatabase<typeof tenantSchema>;
	organizationId: string;
	branchNodeId: string;
	/** Your GHL agency / company id (constant for the whole agency) */
	companyId: string;
	user: AvantejUser;
	member: AvantejMember;
	operation: SyncStaffOperation;
	decryptApiKey: (encrypted: string) => string;
}

export interface SyncStaffResult {
	status: "synced" | "skipped" | "error";
	ghlUserId?: string;
	error?: string;
	reason?: "branch-not-connected" | "non-staff-role";
}

export async function syncStaffToGhl(
	args: SyncStaffArgs,
): Promise<SyncStaffResult> {
	const {
		controlPanelDb,
		tenantDb,
		organizationId,
		branchNodeId,
		companyId,
		user,
		member,
		operation,
		decryptApiKey,
	} = args;

	if (
		member.role !== "admin" &&
		member.role !== "staff" &&
		member.role !== "trainer"
	) {
		return { status: "skipped", reason: "non-staff-role" };
	}

	const link = await controlPanelDb.query.ghlBranchLinks.findFirst({
		where: and(
			eq(ghlBranchLinks.organizationId, organizationId),
			eq(ghlBranchLinks.tenantBranchId, branchNodeId),
			eq(ghlBranchLinks.isActive, true),
		),
	});
	if (!link) return { status: "skipped", reason: "branch-not-connected" };

	const apiKey = decryptApiKey(link.ghlApiKey);
	const locationId = link.ghlLocationId;

	const existing = await tenantDb.query.ghlUserLinks.findFirst({
		where: and(
			eq(ghlUserLinks.memberId, member.id),
			eq(ghlUserLinks.branchNodeId, branchNodeId),
		),
	});

	try {
		if (operation === "create" || (operation === "update" && !existing)) {
			const payload = buildUserPayload({
				companyId,
				locationId,
				user,
				memberRole: member.role as "admin" | "staff" | "trainer",
			});
			const created = await ghlFetch<GhlUser>({
				method: "POST",
				path: "/users/",
				apiKey,
				body: payload,
			});
			await upsertUserLink({
				tenantDb,
				ghlUserId: created.id,
				userId: user.id,
				memberId: member.id,
				branchNodeId,
				ghlLocationId: locationId,
				syncStatus: "active",
				lastError: null,
			});
			return { status: "synced", ghlUserId: created.id };
		}

		if (operation === "update") {
			const payload = buildUserPayload({
				companyId,
				locationId,
				user,
				memberRole: member.role as "admin" | "staff" | "trainer",
			});
			await ghlFetch<GhlUser>({
				method: "PUT",
				path: `/users/${existing!.ghlUserId}`,
				apiKey,
				body: payload,
			});
			await upsertUserLink({
				tenantDb,
				ghlUserId: existing!.ghlUserId,
				userId: user.id,
				memberId: member.id,
				branchNodeId,
				ghlLocationId: locationId,
				syncStatus: "active",
				lastError: null,
			});
			return { status: "synced", ghlUserId: existing!.ghlUserId };
		}

		// delete
		if (!existing) return { status: "skipped" };
		await ghlFetch<unknown>({
			method: "DELETE",
			path: `/users/${existing.ghlUserId}`,
			apiKey,
		});
		await upsertUserLink({
			tenantDb,
			ghlUserId: existing.ghlUserId,
			userId: user.id,
			memberId: member.id,
			branchNodeId,
			ghlLocationId: locationId,
			syncStatus: "deleted",
			lastError: null,
		});
		return { status: "synced", ghlUserId: existing.ghlUserId };
	} catch (error) {
		const message =
			error instanceof GhlApiError
				? `${error.code}: ${error.message}`
				: error instanceof Error
					? error.message
					: String(error);
		await upsertUserLink({
			tenantDb,
			ghlUserId: existing?.ghlUserId ?? "pending",
			userId: user.id,
			memberId: member.id,
			branchNodeId,
			ghlLocationId: locationId,
			syncStatus: "error",
			lastError: message,
		});
		return { status: "error", error: message };
	}
}

async function upsertUserLink(args: {
	tenantDb: NeonDatabase<typeof tenantSchema>;
	ghlUserId: string;
	userId: string;
	memberId: string;
	branchNodeId: string;
	ghlLocationId: string;
	syncStatus: "active" | "deleted" | "error";
	lastError: string | null;
}) {
	await args.tenantDb
		.insert(ghlUserLinks)
		.values({
			ghlUserId: args.ghlUserId,
			userId: args.userId,
			memberId: args.memberId,
			branchNodeId: args.branchNodeId,
			ghlLocationId: args.ghlLocationId,
			syncStatus: args.syncStatus,
			lastSyncedAt: new Date(),
			lastError: args.lastError,
		})
		.onConflictDoUpdate({
			target: [ghlUserLinks.memberId, ghlUserLinks.branchNodeId],
			set: {
				ghlUserId: args.ghlUserId,
				ghlLocationId: args.ghlLocationId,
				syncStatus: args.syncStatus,
				lastSyncedAt: new Date(),
				lastError: args.lastError,
				updatedAt: new Date(),
			},
		});
}

	connectBranch: protectedProcedure
		.input(
			z.object({
				organizationId: z.string().uuid(),
				tenantBranchId: z.string().uuid(),
				ghlLocationId: z.string().min(1),
				ghlApiKey: z.string().min(10),
			}),
		)
		.handler(async ({ input, context }) => {
			const controlPanelDb = context.controlPanelDb;
			if (!controlPanelDb) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Control panel DB not available",
				});
			}

			// 1. Validate the API key works for the locationId
			const valid = await validateGhlApiKey(input.ghlApiKey, input.ghlLocationId);
			if (!valid) {
				throw new ORPCError("BAD_REQUEST", {
					message:
						"GHL API key is invalid or does not belong to that locationId. Double-check the values.",
				});
			}

			// 2. Bootstrap custom fields (idempotent)
			try {
				await ensureCustomFieldsExist({
					apiKey: input.ghlApiKey,
					locationId: input.ghlLocationId,
				});
			} catch (error) {
				// Non-fatal — connect anyway, but warn.
				console.warn("Custom field bootstrap failed", error);
			}

			// 3. Encrypt + upsert the link
			const encryptedKey = encrypt(input.ghlApiKey);
			await controlPanelDb
				.insert(ghlBranchLinks)
				.values({
					organizationId: input.organizationId,
					tenantBranchId: input.tenantBranchId,
					ghlLocationId: input.ghlLocationId,
					ghlApiKey: encryptedKey,
					isActive: true,
				})
				.onConflictDoUpdate({
					target: [
						ghlBranchLinks.organizationId,
						ghlBranchLinks.tenantBranchId,
					],
					set: {
						ghlLocationId: input.ghlLocationId,
						ghlApiKey: encryptedKey,
						isActive: true,
						updatedAt: new Date(),
					},
				});
			return { status: "connected" as const };
		}),

	disconnectBranch: protectedProcedure
		.input(
			z.object({
				organizationId: z.string().uuid(),
				tenantBranchId: z.string().uuid(),
			}),
		)
		.handler(async ({ input, context }) => {
			const controlPanelDb = context.controlPanelDb;
			if (!controlPanelDb) {
				throw new ORPCError("INTERNAL_SERVER_ERROR", {
					message: "Control panel DB not available",
				});
			}
			await controlPanelDb
				.update(ghlBranchLinks)
				.set({ isActive: false, updatedAt: new Date() })
				.where(
					and(
						eq(ghlBranchLinks.organizationId, input.organizationId),
						eq(ghlBranchLinks.tenantBranchId, input.tenantBranchId),
					),
				);
			return { status: "disconnected" as const };
		}),
};
```
